function diagnoseParticipantsHeader() {
  const ss = SpreadsheetApp.getActiveSpreadsheet();
  const sheet = Participants.getParticipantsSheet(ss);
  const lastCol = sheet.getLastColumn();
  const header = sheet.getRange(1, 1, 1, lastCol).getValues()[0];

  const message = `Колонок в заголовке: ${lastCol}\n\n` +
    header.map((h, i) => `${i + 1}. ${h}`).join("\n");

  Logger.log(message);
  SpreadsheetApp.getUi().alert("Диагностика: Участники", message, SpreadsheetApp.getUi().ButtonSet.OK);
}
